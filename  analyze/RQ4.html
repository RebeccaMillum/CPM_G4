plot_Response_boxplot <- function(data, variable, y_label, test_result) {
  ggplot(data, aes(x = Response, y = .data[[variable]])) +
    geom_boxplot(outlier.shape = NA) +
    geom_jitter(width = 0.12, alpha = 0.6) +
    annotate(
      "text",
      x = Inf, y = Inf,
      label = paste0("p = ", signif(test_result$p.value, 3)),
      hjust = 1.1, vjust = 1.5
    ) +
    labs(
      x = "Response class",
      y = y_label
    ) +
    theme_minimal()
}

plot_association <- function(data, x, y, x_label, y_label, cor_test) {
  rho <- unname(cor_test$estimate)

  ggplot(data, aes(x = .data[[x]], y = .data[[y]])) +
    geom_point(alpha = 0.7) +
    geom_smooth(method = "lm", se = FALSE) +
    annotate(
      "text",
      x = Inf, y = Inf,
      label = paste0(
        "Spearman rho = ", round(rho, 2),
        "\np = ", signif(cor_test$p.value, 3)
      ),
      hjust = 1.1, vjust = 1.3
    ) +
    labs(
      x = x_label,
      y = y_label
    ) +
    theme_minimal()
}
